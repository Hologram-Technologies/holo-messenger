\begin{lexlean}{Messenger.Layout}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: panes,
  parameters: [width : Nat],
  result: Nat,
  body: (if (width <? 768) then 1 else (if (width <? 1200) then 2 else 3))
}

definition{
  name: listVisible,
  parameters: [width : Nat, roomOpen : Nat],
  result: Nat,
  body: (if (width <? 768) then (if (roomOpen =? 0) then 1 else 0) else 1)
}

definition{
  name: timelineVisible,
  parameters: [width : Nat, roomOpen : Nat],
  result: Nat,
  body: (if (width <? 768) then (if (roomOpen =? 0) then 0 else 1) else 1)
}

definition{
  name: rightPanelMode,
  parameters: [width : Nat, isOpen : Nat],
  result: Nat,
  body: (if (isOpen =? 0) then 0 else (if (width <? 1200) then 1 else 2))
}

definition{
  name: bottomTabs,
  parameters: [width : Nat],
  result: Bool,
  body: (width <? 768)
}

definition{
  name: listed,
  parameters: [isSpace : Nat, isPlaceholder : Nat],
  result: Nat,
  body: (if (isSpace =? 0) then (if (isPlaceholder =? 0) then 1 else 0) else 0)
}

theorem{
  name: atLeastOnePane,
  parameters: [width : Nat],
  statement: ((1 <=? (panes width)) = true),
  proof: linear_arithmetic{definitions: [panes]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: atMostThreePanes,
  parameters: [width : Nat],
  statement: (((panes width) <=? 3) = true),
  proof: linear_arithmetic{definitions: [panes]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: narrowestPhoneHasOnePane,
  parameters: [],
  statement: ((panes 320) = 1),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: phoneNeverShowsListAndTimeline,
  parameters: [width : Nat, roomOpen : Nat],
  statement: (((width <? 768) && (2 <=? ((listVisible width roomOpen) + (timelineVisible width roomOpen)))) = false),
  proof: linear_arithmetic{definitions: [listVisible, timelineVisible]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: alwaysShowsListOrTimeline,
  parameters: [width : Nat, roomOpen : Nat],
  statement: ((1 <=? ((listVisible width roomOpen) + (timelineVisible width roomOpen))) = true),
  proof: linear_arithmetic{definitions: [listVisible, timelineVisible]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: closedPanelHidden,
  parameters: [width : Nat],
  statement: ((rightPanelMode width 0) = 0),
  proof: linear_arithmetic{definitions: [rightPanelMode]},
  axioms: [propext]
}

theorem{
  name: dockedPanelNeedsWidth,
  parameters: [width : Nat, isOpen : Nat],
  statement: ((((rightPanelMode width isOpen) =? 2) && (width <? 1200)) = false),
  proof: linear_arithmetic{definitions: [rightPanelMode]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: spacesNeverListed,
  parameters: [isPlaceholder : Nat],
  statement: ((listed 1 isPlaceholder) = 0),
  proof: linear_arithmetic{definitions: [listed]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: placeholdersNeverListed,
  parameters: [isSpace : Nat],
  statement: ((listed isSpace 1) = 0),
  proof: linear_arithmetic{definitions: [listed]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
