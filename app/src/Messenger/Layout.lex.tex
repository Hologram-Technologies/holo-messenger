\begin{semanticmodule}
definition{name: panelHidden, parameters: [], result: Nat, body: 0}

definition{name: panelOverlay, parameters: [], result: Nat, body: 1}

definition{name: panelDocked, parameters: [], result: Nat, body: 2}

definition{
  name: panes,
  parameters: [width : Nat],
  result: Nat,
  body: (if (width <? 768) then 1 else (if (width <? 1200) then 2 else 3))
}

  name: listVisible,
  parameters: [width : Nat, roomOpen : Nat],
  result: Nat,
  body: (if (width <? 768) then (if (roomOpen =? 0) then 1 else 0) else 1)
}

definition{
  name: timelineVisible,
  parameters: [width : Nat, roomOpen : Nat],
  result: Nat,
  body: (if (width <? 768) then (if (roomOpen =? 0) then 0 else 1) else 1)
}

definition{
  name: rightPanelMode,
  parameters: [width : Nat, isOpen : Nat],
  result: Nat,
  body: (if (isOpen =? 0) then 0 else (if (width <? 1200) then 1 else 2))
}

definition{name: bottomTabs, parameters: [width : Nat], result: Bool, body: (width <? 768)}

definition{
  name: listed,
  parameters: [isSpace : Nat, isPlaceholder : Nat],
  result: Nat,
  body: (if (isSpace =? 0) then (if (isPlaceholder =? 0) then 1 else 0) else 0)
}
