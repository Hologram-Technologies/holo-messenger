\begin{lexlean}{Messenger.Composer}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: enterAction,
  parameters: [shift : Nat, modifier : Nat, requireModifier : Nat, popupOpen : Nat],
  result: Nat,
  body: (if (1 <=? popupOpen) then 3 else (if (1 <=? shift) then 2 else (if (requireModifier =? 0) then 1 else (if (1 <=? modifier) then 1 else 2))))
}

definition{
  name: escapeStep,
  parameters: [mode : Nat, popupOpen : Nat],
  result: Nat,
  body: (if (1 <=? popupOpen) then mode else 0)
}

definition{
  name: upEditsPrevious,
  parameters: [empty : Nat, popupOpen : Nat, hasOwnMessage : Nat],
  result: Nat,
  body: (if (empty =? 0) then 0 else (if (1 <=? popupOpen) then 0 else (if (1 <=? hasOwnMessage) then 1 else 0)))
}

definition{
  name: sendable,
  parameters: [length : Nat, attachments : Nat],
  result: Bool,
  body: (1 <=? (length + attachments))
}

definition{
  name: draftKept,
  parameters: [length : Nat, mode : Nat],
  result: Bool,
  body: (1 <=? (length + mode))
}

theorem{
  name: shiftEnterNeverSends,
  parameters: [modifier : Nat, requireModifier : Nat, popupOpen : Nat],
  statement: (((enterAction 1 modifier requireModifier popupOpen) =? 1) = false),
  proof: linear_arithmetic{definitions: [enterAction]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: openPopupEnterCompletes,
  parameters: [shift : Nat, modifier : Nat, requireModifier : Nat],
  statement: ((enterAction shift modifier requireModifier 1) = 3),
  proof: linear_arithmetic{definitions: [enterAction]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: plainEnterWaitsForModifier,
  parameters: [],
  statement: ((enterAction 0 0 1 0) = 2),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: plainEnterSendsByDefault,
  parameters: [],
  statement: ((enterAction 0 0 0 0) = 1),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: escapeClosesPopupFirst,
  parameters: [mode : Nat],
  statement: ((escapeStep mode 1) = mode),
  proof: linear_arithmetic{definitions: [escapeStep]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: escapeCancelsReplyOrEdit,
  parameters: [mode : Nat],
  statement: ((escapeStep mode 0) = 0),
  proof: linear_arithmetic{definitions: [escapeStep]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: typedTextBlocksUpEdit,
  parameters: [popupOpen : Nat, hasOwnMessage : Nat],
  statement: ((upEditsPrevious 0 popupOpen hasOwnMessage) = 0),
  proof: linear_arithmetic{definitions: [upEditsPrevious]},
  axioms: [propext]
}

theorem{
  name: emptyNeverSends,
  parameters: [],
  statement: ((sendable 0 0) = false),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: emptyPlainDraftDropped,
  parameters: [],
  statement: ((draftKept 0 0) = false),
  proof: reflexivity{},
  axioms: []
}
\end{semanticmodule}
\end{lexlean}
