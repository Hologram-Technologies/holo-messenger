definition{
  name: modalNone,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: modalNormal,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: modalFixed,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: modalUrgent,
  parameters: [],
  result: Nat,
  body: 3
}

definition{
  name: modalShown,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  result: Nat,
  body: (if (1 <=? urgent) then 3 else (if (1 <=? fixed) then 2 else (if (1 <=? normal) then 1 else 0)))
}

definition{
  name: focusTrapped,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  result: Nat,
  body: (if (1 <=? (fixed + (urgent + normal))) then 1 else 0)
}

definition{
  name: escapeCloses,
  parameters: [menuOpen : Nat, modalKind : Nat],
  result: Nat,
  body: (if (1 <=? menuOpen) then 1 else (if (modalKind =? 0) then 0 else (if (modalKind =? 2) then 0 else 2)))
}

definition{
  name: topToast,
  parameters: [x : Nat, y : Nat],
  result: Nat,
  body: (if (x <? y) then y else x)
}
