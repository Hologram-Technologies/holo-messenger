\begin{lexlean}{Messenger.Overlay}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: modalNone,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: modalNormal,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: modalFixed,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: modalUrgent,
  parameters: [],
  result: Nat,
  body: 3
}

definition{
  name: modalShown,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  result: Nat,
  body: (if (1 <=? urgent) then 3 else (if (1 <=? fixed) then 2 else (if (1 <=? normal) then 1 else 0)))
}

definition{
  name: focusTrapped,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  result: Nat,
  body: (if (1 <=? (fixed + (urgent + normal))) then 1 else 0)
}

definition{
  name: escapeCloses,
  parameters: [menuOpen : Nat, modalKind : Nat],
  result: Nat,
  body: (if (1 <=? menuOpen) then 1 else (if (modalKind =? 0) then 0 else (if (modalKind =? 2) then 0 else 2)))
}

definition{
  name: topToast,
  parameters: [x : Nat, y : Nat],
  result: Nat,
  body: (if (x <? y) then y else x)
}

theorem{
  name: urgentModalAlwaysWins,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  statement: (((1 <=? urgent) && (((modalShown fixed urgent normal) <? 3) || (3 <? (modalShown fixed urgent normal)))) = false),
  proof: linear_arithmetic{definitions: [modalShown]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: modalInRange,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  statement: (((modalShown fixed urgent normal) <=? 3) = true),
  proof: linear_arithmetic{definitions: [modalShown]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: focusTrappedWheneverShown,
  parameters: [fixed : Nat, urgent : Nat, normal : Nat],
  statement: (((1 <=? (modalShown fixed urgent normal)) && ((focusTrapped fixed urgent normal) =? 0)) = false),
  proof: linear_arithmetic{definitions: [focusTrapped, modalShown]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: menuClosesBeforeModal,
  parameters: [menuOpen : Nat, modalKind : Nat],
  statement: (((1 <=? menuOpen) && (((escapeCloses menuOpen modalKind) <? 1) || (1 <? (escapeCloses menuOpen modalKind)))) = false),
  proof: linear_arithmetic{definitions: [escapeCloses]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: overlayExamples,
  parameters: [],
  statement: ((((modalShown 0 0 0) =? 0) && (((escapeCloses 0 (modalFixed)) =? 0) && ((escapeCloses 0 (modalNormal)) =? 2))) = true),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: topToastCoversLeft,
  parameters: [x : Nat, y : Nat],
  statement: ((x <=? (topToast x y)) = true),
  proof: linear_arithmetic{definitions: [topToast]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: topToastCoversRight,
  parameters: [x : Nat, y : Nat],
  statement: ((y <=? (topToast x y)) = true),
  proof: linear_arithmetic{definitions: [topToast]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
