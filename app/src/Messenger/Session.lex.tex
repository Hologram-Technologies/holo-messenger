definition{
  name: rootView,
  parameters: [supported : Nat, started : Nat, hasIdentity : Nat, holdsLock : Nat],
  result: Nat,
  body: (if (supported =? 0) then 4 else (if (started =? 0) then 0 else (if (hasIdentity =? 0) then 1 else (if (holdsLock =? 0) then 3 else 2))))
}

definition{
  name: lockStep,
  parameters: [state : Nat, signal : Nat],
  result: Nat,
  body: (if (3 <=? state) then 3 else (if (signal =? 3) then (if (state =? 1) then 3 else state) else (if (state =? 0) then (if (signal =? 0) then 1 else (if (signal =? 1) then 2 else 0)) else (if (state =? 2) then (if (signal =? 2) then 1 else (if (signal =? 4) then 0 else 2)) else 1))))
}

definition{
  name: wipeAllowed,
  parameters: [confirmed : Nat],
  result: Bool,
  body: (1 <=? confirmed)
}

theorem{
  name: mainNeedsIdentity,

theorem{
