\title{Boolean}
\begin{semanticmodule}
definition{
  name: resolvesDark,
  parameters: [choice : Nat, systemDark : Nat],
  result: Nat,
  body: (if (choice =? 0) then (if (1 <=? systemDark) then 1 else 0) else (if (2 <=? choice) then 1 else 0))
}

definition{
  name: fontSize,
  parameters: [requested : Nat],
  result: Nat,
  body: (if (requested <? 12) then 12 else (if (24 <? requested) then 24 else requested))
}

definition{
  name: changeStep,
  parameters: [state : Nat, signal : Nat],
  result: Nat,
  body: (if (signal =? 0) then 1 else (if (state =? 1) then (if (signal =? 1) then 2 else (if (signal =? 2) then 3 else 1)) else (if (3 <? state) then 0 else state)))
}

definition{
  name: shownValue,
  parameters: [state : Nat, oldValue : Nat, newValue : Nat],
  result: Nat,
  body: (if (state =? 1) then newValue else (if (state =? 2) then newValue else oldValue))
}

definition{
  name: activates,
  parameters: [verified : Nat, reopened : Nat],
  result: Nat,
  body: (if (verified =? 0) then 0 else (if (1 <=? reopened) then 1 else 0))
}

theorem{
  name: lightIgnoresSystem,
  parameters: [systemDark : Nat],
  statement: ((resolvesDark 1 systemDark) = 0),
  proof: linear_arithmetic{definitions: [resolvesDark]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: darkIgnoresSystem,
  parameters: [systemDark : Nat],
  statement: ((resolvesDark 2 systemDark) = 1),
  proof: linear_arithmetic{definitions: [resolvesDark]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: fontNeverBelowMinimum,
  parameters: [requested : Nat],
  statement: ((12 <=? (fontSize requested)) = true),
  proof: linear_arithmetic{definitions: [fontSize]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: fontNeverAboveMaximum,

  proof: linear_arithmetic{definitions: [fontSize]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: failedChangeShowsOldValue,
  parameters: [oldValue : Nat, newValue : Nat],

  proof: linear_arithmetic{definitions: [shownValue]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: pendingChangeShowsNewValue,

  axioms: [propext]
}

  name: savedOnlyAfterSaving,

  name: unverifiedUpdateNeverActivates,
