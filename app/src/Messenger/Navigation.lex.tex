definition{
  name: goForward,
  parameters: [position : Nat, length : Nat],
  result: Nat,
  body: (if ((position + 1) <? length) then (position + 1) else position)
}

definition{
  name: routeTarget,
  parameters: [kind : Nat, known : Nat, status : Nat],
  result: Nat,
  body: (if (known =? 0) then 0 else (if (kind =? 2) then 3 else (if (kind =? 3) then 4 else (if (4 <? kind) then 0 else (if (status =? 2) then 1 else 2)))))
